services:
  qdrant:
    image: qdrant/qdrant:latest
    container_name: qdrant
    restart: unless-stopped

    ports:
      - "6333:6333"   # REST API + Web UI
      - "6334:6334"   # gRPC API

    volumes:
      - /volume1/docker/qdrant/storage:/qdrant/storage
      - /volume1/docker/qdrant/snapshots:/qdrant/snapshots
      - /volume1/docker/qdrant/config:/qdrant/config

    environment:
      - QDRANT__SERVICE__HTTP_PORT=6333
      - QDRANT__SERVICE__GRPC_PORT=6334

      # Uncomment to enable API key auth (recommended if exposed on your LAN)
      # - QDRANT__SERVICE__API_KEY=your_secret_key_here

      # Uncomment to enable read-only key for query-only clients
      # - QDRANT__SERVICE__READ_ONLY_API_KEY=your_readonly_key_here

      # Storage tuning — lower RAM usage on NAS hardware
      - QDRANT__STORAGE__PERFORMANCE__MAX_SEARCH_THREADS=2
      - QDRANT__STORAGE__OPTIMIZERS__DEFAULT_SEGMENT_NUMBER=2
      - QDRANT__STORAGE__OPTIMIZERS__MEMMAP_THRESHOLD_KB=20000

    # Optional: cap resources to avoid starving other NAS services
    deploy:
      resources:
        limits:
          memory: 4G
          cpus: "2.0"

    healthcheck:
      test: ["CMD", "curl", "-f", "http://localhost:6333/readyz"]
      interval: 30s
      timeout: 10s
      retries: 3
      start_period: 20s
